cmake_minimum_required(VERSION 3.16)
project(PchTest C CXX)

option(UNITTESTS "Build test PCH" ON)
option(CMAKE_POSITION_INDEPENDENT_CODE "Build position independent code" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# These flags were missing when the donor was created in contrib/libs.
if (CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
	set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -D_BSD_SOURCE -D_DEFAULT_SOURCE -Werror=invalid-pch")
	set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -D_BSD_SOURCE -D_DEFAULT_SOURCE -Werror=invalid-pch")
endif()
get_filename_component(ENGINE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include(${ENGINE_SOURCE_DIR}/cmake/macros.cmake)
add_library(sdl3 INTERFACE)
target_include_directories(sdl3 INTERFACE ${ENGINE_SOURCE_DIR}/contrib/libs/sdl3/include)
add_library(gtest INTERFACE)
target_include_directories(gtest INTERFACE ${ENGINE_SOURCE_DIR}/contrib/libs/gtest/include)
include(${ENGINE_SOURCE_DIR}/cmake/pch.cmake)

if (MSVC AND NOT CMAKE_DISABLE_PRECOMPILE_HEADERS)
	get_target_property(_donor_type sdl3-pch TYPE)
	if (NOT _donor_type STREQUAL "STATIC_LIBRARY")
		message(FATAL_ERROR "MSVC PCH donor must be STATIC so cmake_pch.obj is linkable")
	endif()
endif()

foreach(_target first second application)
	if (_target STREQUAL "application")
		add_executable(${_target} main.cpp shared.c)
		set(_donor sdl3-pch-executable)
	else()
		add_library(${_target} STATIC shared.cpp shared.c)
		set(_donor sdl3-pch)
	endif()
	target_link_libraries(${_target} PRIVATE sdl3)
	target_compile_definitions(${_target} PRIVATE SDL_MAIN_HANDLED)
	engine_reuse_sdl3_pch(${_target})
	get_target_property(_reuse ${_target} PRECOMPILE_HEADERS_REUSE_FROM)
	if (CMAKE_DISABLE_PRECOMPILE_HEADERS)
		if (_reuse OR TARGET ${_donor})
			message(FATAL_ERROR "PCH must be disabled for ${_target}")
		endif()
	elseif (NOT _reuse STREQUAL _donor)
		message(FATAL_ERROR "${_target} must reuse ${_donor}")
	endif()
endforeach()

# Target-specific optimization must not reuse a PCH with different flags.
add_library(optimized STATIC shared.cpp shared.c)
target_link_libraries(optimized PRIVATE sdl3)
target_compile_definitions(optimized PRIVATE SDL_MAIN_HANDLED)
engine_reuse_sdl3_pch(optimized)
engine_target_optimize(optimized)

# Mixed C/OBJC libraries must still reuse the shared donor (app/AppDocument.m).
if (APPLE)
	enable_language(OBJC)
	add_library(objc_lib STATIC shared.cpp shared.c shared.m)
	target_link_libraries(objc_lib PRIVATE sdl3)
	target_compile_definitions(objc_lib PRIVATE SDL_MAIN_HANDLED)
	engine_reuse_sdl3_pch(objc_lib)
	get_target_property(_reuse objc_lib PRECOMPILE_HEADERS_REUSE_FROM)
	if (CMAKE_DISABLE_PRECOMPILE_HEADERS)
		if (_reuse)
			message(FATAL_ERROR "PCH must be disabled for objc_lib")
		endif()
	elseif (NOT _reuse STREQUAL "sdl3-pch")
		message(FATAL_ERROR "objc_lib must reuse sdl3-pch")
	endif()
endif()

if (UNITTESTS)
	add_executable(suite suite.cpp)
	target_link_libraries(suite PRIVATE gtest sdl3)
	target_compile_definitions(suite PRIVATE SDL_MAIN_HANDLED)
	target_compile_options(suite PRIVATE $<$<CXX_COMPILER_ID:GNU>:-Wno-undef>)
	if (NOT CMAKE_DISABLE_PRECOMPILE_HEADERS)
		target_precompile_headers(suite REUSE_FROM gtest-pch)
	endif()
elseif (TARGET gtest-pch)
	message(FATAL_ERROR "Test PCH must be disabled with UNITTESTS=OFF")
endif()
